Express Backend API Client Snippet

Discover healthy dynamic microservice instances in Node.js before forwarding client traffic.

service-client.js (Express & consul-node) ESM / Async Await 
// Query Consul health catalog for passing instances only
const consul = require('consul')({ host: '127.0.0.1', port: 8500 });

app.get('/api/checkout', async (req, res) => {
  try {
    // Dynamic discovery filter ensures no unhealthy instances are routed to
    const instances = await consul.health.service({
      service: 'payment-service',
      passing: true
    });

    if (!instances.length) {
      return res.status(503).json({ error: 'No healthy payment nodes resolved' });
    }

    // Format destination using dynamic Consul host:port resolution
    const targetUrl = `http://${instances[0].Service.Address}:${instances[0].Service.Port}/pay`;
    // Forward request to resolved target...
    res.json({ routedTo: targetUrl, status: 'PROXIED' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});
