Consul Registration REST Endpoint (Express / Node.js)

Production-ready Express.js route registering dynamic workloads into HashiCorp Consul daemon.

server/routes/registry.js
Node.js 20 LTS • consul@1.2.0 
const express = require('express');
const Consul = require('consul');
const router = express.Router();

// Initialize Consul Client connected to local agent
const consul = new Consul({
  host: process.env.CONSUL_HOST || '127.0.0.1',
  port: process.env.CONSUL_PORT || '8500',
  promisify: true
});

/**
 * POST /api/register
 * Handles incoming microservice registration into Consul Catalog
 */
router.post('/api/register', async (req, res) => {
  const { name, id, address, port, checkPath, interval = '10s', tags = [] } = req.body;

  try {
    // Register service with Consul Agent
    await consul.agent.service.register({
      name: name,
      id: id || `${name}-${Date.now()}`,
      address: address,
      port: Number(port),
      tags: tags,
      check: {
        http: `http://${address}:${port}${checkPath}`,
        interval: interval,
        timeout: '2s',
        deregistercriticalserviceafter: '1m'
      }
    });

    return res.status(201).json({
      success: true,
      message: `Service ${name} [${id}] registered with Consul`,
      consulCatalogUrl: `http://127.0.0.1:8500/v1/catalog/service/${name}`
    });
  } catch (err) {
    return res.status(500).json({
      success: false,
      error: err.message
    });
  }
});
