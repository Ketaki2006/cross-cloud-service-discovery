Express.js Integration

Node.js API

Sample backend route demonstrating how your Express gateway pulls only healthy, passing services from Consul:

routes/health.js 
const express = require('express');
const axios = require('axios');
const router = express.Router();

// Query passing instances for active mesh routing
router.get('/api/health/:service', async (req, res) => {
  try {
    const { service } = req.params;
    const consulRes = await axios.get(
      `http://localhost:8500/v1/health/service/${service}?passing=true`
    );
    return res.json({
      status: 'healthy',
      healthyNodes: consulRes.data.length,
      instances: consulRes.data.map(i => ({
        id: i.Service.ID,
        address: i.Service.Address,
        port: i.Service.Port
      }))
    });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});
